Comments :: [].{
	wrapped_with_comments : I64 -> I64
	wrapped_with_comments = |x| # the opening's comment
		x + 1 # the closing line's comment

	condition_comment : I64 -> I64
	condition_comment = |x|
		if x > 1 { # after the condition
			2
		} else { # after else
			3
		}

	both_findings : I64 -> I64
	both_findings = |x|
		if x > 1 {
			2
		} else {
			3
		}

	block_statement : I64 -> I64
	block_statement = |x| {
		x + 1
	}

	else_block_with_comment : I64 -> I64
	else_block_with_comment = |x|
		if x > 1 {
			2
		} else # the block's comment
			if x > 0 {
				1
			} else {
				0
			}

	unicode_before_the_if : Str -> I64
	unicode_before_the_if = |s|
		if s == "é" {
			1
		} else {
			2
		}

	unary_condition : Bool -> I64
	unary_condition = |b|
		if !b {
			1
		} else {
			2
		}

	unary_branches : Bool -> Bool
	unary_branches = |b|
		if b {
			!b
		} else {
			!(!b)
		}

	match_in_parens : I64 -> I64
	match_in_parens = |x|
		match x {
			0 => 1
			_ => 2
		}

	nested_wrapped : I64 -> I64
	nested_wrapped = |x| x + 1

	branch_body_block : [A, B] -> I64
	branch_body_block = |t|
		match t {
			A => {
				y = 1
				y + 1
			}
			B => 3
		}

	declaration_over_lines : I64 -> I64
	declaration_over_lines = |x| {
		y = x
			+ 1
		y
	}

	else_if_braceless : I64 -> I64
	else_if_braceless = |x|
		if x > 1 {
			2
		} else if x > 0 {
			1
		} else {
			0
		}

	else_block_in_else_block : I64 -> I64
	else_block_in_else_block = |x|
		if x > 2 {
			3
		} else if x > 1 {
			2
		} else if x > 0 {
			1
		} else {
			0
		}
}

--- emitted.roc
Emitted :: [].{
	one_line : I64 -> I64
	one_line = |x|
		if x > 1 {
			2
		} else {
			3
		} # paren_if

	whole_condition : I64 -> I64
	whole_condition = |x|
		if x > 1 { # paren_if
			2
		} else {
			3
		}

	wrapped : I64 -> I64
	wrapped = |x| # wrapped_body
		x + 1

	wrapped_on_one_line : I64 -> Bool
	wrapped_on_one_line = |x| x > 1 # wrapped_body

	wrapped_value : I64 -> I64
	wrapped_value = |x| {
		y = # wrapped_body
			if x > 1 {
				2
			} else {
				3
			}
		y + 1 # wrapped_body
	}

	wrapped_branch : [A, B] -> I64
	wrapped_branch = |t|
		match t {
			A => # wrapped_body
				1 + 2
			B => 3
		}

	nested_else : I64 -> I64
	nested_else = |x|
		if x > 1 { # else_block
			2
		} else if x > 0 {
			1
		} else {
			0
		}
}
--- lambda_if.roc
LambdaIf :: [].{
	hanging : I64 -> I64
	hanging = |x|
		if x > 1 { # lambda_if
			2
		} else {
			3
		}

	on_one_line : I64 -> I64
	on_one_line = |x|
		if x > 1 {
			2
		} else {
			3
		} # lambda_if

	else_if_chain : I64 -> I64
	else_if_chain = |x|
		if x > 1 { # lambda_if
			2
		} else if x > 0 {
			1
		} else {
			0
		}

	stacked_arguments : { first : I64, second : I64 } -> I64
	stacked_arguments = |
		{ first, second },
	|
		if first > second { # lambda_if
			first
		} else {
			second
		}

	inside_a_call : List(I64) -> List(I64)
	inside_a_call = |numbers|
		numbers.map(
			|x|
				if x > 1 { # lambda_if
					2
				} else {
					3
				},
		)

	lambda_in_a_lambda : I64 -> (I64 -> I64)
	lambda_in_a_lambda = |x|
		|y|
			if x > y { # lambda_if
				x
			} else {
				y
			}

	braceless_first : I64 -> I64
	braceless_first = |x|
		if x > 1 {
			2
		} else {
			3
		} # paren_if
}
--- quiet.roc
Quiet :: [].{
	braces : I64 -> I64
	braces = |x|
		if x > 1 {
			2
		} else if x > 0 {
			1
		} else {
			0
		}

	grouped_operand : F64, F64 -> F64
	grouped_operand = |a, b| (a * b) + (b * a)

	grouped_condition : F64 -> Bool
	grouped_condition = |a|
		if (a * 2.0) < 1.0 {
			True
		} else {
			False
		}

	one_line_value : F64 -> F64
	one_line_value = |a| {
		doubled = (a * 2.0)
		doubled + 1.0
	}

	grouped_argument : F64 -> F64
	grouped_argument = |a| grouped_operand((a + 1.0), a)

	pair : I64 -> (I64, I64)
	pair = |x| (x, x + 1)

	statements_in_else : I64 -> I64
	statements_in_else = |x|
		if x > 1 {
			2
		} else {
			y = x + 1
			if y > 0 {
				1
			} else {
				0
			}
		}

	match_in_braces : I64, I64 -> I64
	match_in_braces = |x, y|
		if x > 1 {
			match y {
				0 => 1
				_ => 2
			}
		} else {
			3
		}

	if_on_its_own_line : I64 -> I64
	if_on_its_own_line = |x|
		if x > 1 {
			2
		} else {
			3
		}

	if_after_a_comment : I64 -> I64
	if_after_a_comment = |x| # the comment ends the lambda's line
		if x > 1 {
			2
		} else {
			3
		}

	if_on_its_own_line_inside_a_call : List(I64) -> List(I64)
	if_on_its_own_line_inside_a_call = |numbers|
		numbers.map(
			|x|
				if x > 1 {
					2
				} else {
					3
				},
		)

	if_in_a_lambdas_block : I64 -> I64
	if_in_a_lambdas_block = |x| {
		y = x + 1
		if y > 1 {
			2
		} else {
			3
		}
	}

	if_as_an_argument : I64 -> I64
	if_as_an_argument = |x| Quiet.braces(if x > 1 { 2 } else { 3 })

	if_as_an_operand : I64 -> I64
	if_as_an_operand = |x| 1 + (if x > 1 { 2 } else { 3 })

	if_beside_a_declaration : I64 -> I64
	if_beside_a_declaration = |x| {
		y = if x > 1 { 2 } else { 3 }
		y + 1
	}

	match_hanging_off_a_lambda : I64 -> I64
	match_hanging_off_a_lambda = |x| match x {
		0 => 1
		_ => 2
	}
}
